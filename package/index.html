    def T_matrix(D:int) -> Matrix:
    """Return the matrix representation of T in Hermite function to degree D."""
    T = zeros(D+1, D+1)
    for i in range(D+1):
        T[i, i] = 2*i + 1
    for i in range(D-1):
        T[i, i+2] = -sqrt((i+1)*(i+2))
        T[i+2, i] = -sqrt((i+1)*(i+2))
    T /= 4
    return T

    def hermfpmul(f:Sequence, g:Sequence) -> Matrix:
    """Return $m$, such that $mh_0=fg$.

    $f$, $g$ & $m$ are sequences of coefficients
    representing Hermite function series.
    """
    r = [0] * (len(f)+len(g)-1)
    for i, fi in enumerate(f):
        for j, gj in enumerate(g):
            for k in range(min(i, j)+1):
                r[i+j-2*k] += fi * gj \
                        * factorial(k)*comb(i, k)*comb(j, k) \
                        * sqrt(Rational(factorial(i+j-2*k),
                                       (factorial(i)*factorial(j))))
    return Matrix(r)

    def hermf(f:Iterable, x:npt.ArrayLike) -> np.ndarray|np.generic:
    """Return the evaluation of Hermite function series `f` at point `x`."""
    return np.exp(-x**2/2) * np.polynomial.hermite.hermval(
        x,
        [fj / np.sqrt(2**j * factorial(j) * np.sqrt(np.pi))
            for j, fj in enumerate(f)]
    )

    def hermfpmul(f:Sequence, g:Sequence) -> npt.NDArray[np.float64]:
    """Return $m$, such that $mh_0=fg$.

    $f$, $g$ & $m$ are sequences of coefficients
    representing Hermite function series.
    """
    r = np.zeros(len(f)+len(g)-1, dtype=np.float64)
    for i, fi in enumerate(f):
        for j, gj in enumerate(g):
            for k in range(min(i, j)+1):
                r[i+j-2*k] += fi * gj \
                        * factorial(k)*comb(i, k)*comb(j, k) \
                        * np.sqrt(float(factorial(i+j-2*k)) \
                                / (factorial(i)*factorial(j)))
    return r

    def rho_to_g(rho:np.ndarray) -> npt.NDArray[np.float64]:
    r"""Return $\vec{g}$ for density matrix $\rho$."""
    g = np.zeros(rho.shape[0]+rho.shape[1]-1, dtype=np.float64)
    for i in range(rho.shape[0]):
        for j in range(rho.shape[1]):
            hi, hj = np.zeros(i+1, dtype=np.uint8), np.zeros(j+1, dtype=np.uint8)
            hi[i], hj[j] = 1, 1
            hihj = hermfpmul(hi, hj)
            g[:len(hihj)] += rho[i, j] * hihj
    return g
